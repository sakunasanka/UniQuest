CREATE VIEW v_verifiedJobs AS
SELECT 
    j.JobID,
    j.Title,
    j.Email,
    j.Category,
    j.Status,
    v.ActionBy,
    v.ActionDate
FROM 
    v_jobs j
JOIN 
    verificationlogs v
ON 
    j.JobID = v.EntityID
WHERE 
    v.EntityType = 'Job'
    AND v.Action = 'Approve';


CREATE VIEW v_verifiedUsers AS
SELECT 
    u.UserID,
    u.Email,
    u.Role,
    u.Status,
    v.ActionBy,
    v.ActionDate
FROM 
    user u
JOIN 
    verificationlogs v
ON 
    u.UserID = v.EntityID
WHERE 
    v.EntityType = 'User'
    AND v.Action = 'Approve';







DELIMITER $$

CREATE TRIGGER after_user_verified
AFTER UPDATE ON user
FOR EACH ROW
BEGIN
    -- Handle status change to 'Active' (Approval)
    IF NEW.Status = 'Active' AND (OLD.Status = 'Pending' OR OLD.Status = 'Not Approved') THEN
        INSERT INTO verificationlogs 
        (
            EntityID, 
            EntityType, 
            Action, 
            ActionBy, 
            ActionDate, 
            Comments
        ) 
        VALUES 
        (
            NEW.UserID, 
            'User', 
            'Approve', 
            NEW.VerifiedBy, 
            NOW(), 
            'User status changed to Active by admin'
        );
    END IF;

    -- Handle status change to 'Not Approved' (Rejection)
    IF NEW.Status = 'Not Approved' AND OLD.Status = 'Pending' THEN
        INSERT INTO verificationlogs 
        (
            EntityID, 
            EntityType, 
            Action, 
            ActionBy, 
            ActionDate, 
            Comments
        ) 
        VALUES 
        (
            NEW.UserID, 
            'User', 
            'Reject', 
            NEW.VerifiedBy, 
            NOW(), 
            'User status changed to Not Approved by admin'
        );
    END IF;
END$$

DELIMITER ;

-- Creat table for store password_reset Token
CREATE TABLE `password_reset` (
	`UserID` INT(10) NOT NULL,
	`Token` VARCHAR(255) NOT NULL,
	`Expiration` DATETIME NOT NULL,
	PRIMARY KEY (`UserID`, `Token`),
	CONSTRAINT `password_reset` FOREIGN KEY (`UserID`) REFERENCES `user` (`UserID`) ON UPDATE CASCADE ON DELETE CASCADE
);


-- Alter table user to add VerifiedBy foreign key
ALTER TABLE `user` 
ADD CONSTRAINT `fk_verifiedby_user`
    FOREIGN KEY (`VerifiedBy`) 
    REFERENCES `user`(`UserID`) 
    ON DELETE SET NULL 
    ON UPDATE CASCADE;
