CREATE TRIGGER `after_job_verified` AFTER UPDATE ON `jobs` FOR EACH ROW BEGIN
    -- Handle status change to 'Active' (Approval)
    IF NEW.Status = 'Active' AND (OLD.Status = 'Pending' OR OLD.Status = 'Not Approved') THEN
        INSERT INTO verificationlogs 
        (
            EntityID, 
            EntityType, 
            Action, 
            ActionBy, 
            ActionDate, 
            Comment
        ) 
        VALUES 
        (
            NEW.JobID, 
            'Job', 
            'Approve', 
            NEW.VerifiedBy, 
            NOW(), 
            'Job status changed to Active'
        );
    END IF;

    -- Handle status change to 'Not Approved' (Rejection)
    IF NEW.Status = 'Not Approved' AND OLD.Status = 'Pending' THEN
        INSERT INTO verificationlogs 
        (
            EntityID, 
            EntityType, 
            Action, 
            ActionBy, 
            ActionDate, 
            Comment
        ) 
        VALUES 
        (
            NEW.JobID, 
            'Job', 
            'Reject', 
            NEW.VerifiedBy, 
            NOW(), 
            'Job status changed to Not Approved'
        );
    END IF;
END//

-- Removing temporary table and create final VIEW structure
DROP TABLE IF EXISTS `studentjobcomplaints`;
CREATE ALGORITHM=UNDEFINED SQL SECURITY DEFINER VIEW `studentjobcomplaints` AS SELECT
    `uniquest`.`jobs`.`Title` AS `JobTitle`,
    `uniquest`.`user`.`UserID` AS `CompanyID`,
    `uniquest`.`user`.`Email` AS `CompanyEmail`,
    `uniquest`.`company`.`CompanyName` AS `CompanyName`,
    `uniquest`.`complaint_jobs`.`ComplaintID` AS `ComplaintID`,
    `uniquest`.`complaint_jobs`.`Description` AS `Complaint`,
    CONCAT(
        `uniquest`.`student`.`FirstName`,
        ' ',
        `uniquest`.`student`.`LastName`
    ) AS `StudentName`,
    `uniquest`.`complaint_jobs`.`ComplainedDate` AS `ComplainedDate`,
    `uniquest`.`complaint_jobs`.`Status` AS `Status`
FROM
    (
        (
            (
                (
                    `uniquest`.`complaint_jobs`
                JOIN `uniquest`.`jobs` ON
                    (
                        `uniquest`.`complaint_jobs`.`JobID` = `uniquest`.`jobs`.`JobID`
                    )
                )
            JOIN `uniquest`.`company` ON
                (
                    `uniquest`.`jobs`.`CompanyID` = `uniquest`.`company`.`CompanyID`
                )
            )
        JOIN `uniquest`.`user` ON
            (
                `uniquest`.`company`.`CompanyID` = `uniquest`.`user`.`UserID`
            )
        )
    JOIN `uniquest`.`student` ON
        (
            `uniquest`.`complaint_jobs`.`StudentID` = `uniquest`.`student`.`StudentID`
        )
    ) ;
