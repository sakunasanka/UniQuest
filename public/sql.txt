-- View `uniquest`.`StudentCompanyComplaints`
-- -----------------------------------------------------

CREATE VIEW `uniquest`.`StudentCompanyComplaints` AS
SELECT 
    c.CompanyName AS `Company`,
    u.Email AS `CompanyEmail`,
    cm.Complaint AS `Complaint`,
    CONCAT(s.FirstName, ' ', s.LastName) AS `StudentName`,
    cm.ComplainedDate AS `ComplainedDate`,
    cm.Status AS `Status`
FROM 
    `uniquest`.`complaint_company` cm
JOIN 
    `uniquest`.`student` s ON cm.StudentID = s.StudentID
JOIN 
    `uniquest`.`company` c ON cm.CompanyID = c.CompanyID
JOIN 
    `uniquest`.`user` u ON c.CompanyID = u.UserID 
ORDER BY 
    cm.ComplainedDate ASC;

    j.Category,                     -- Job category
    j.JobBenefits,                  -- Job benefits

    j.RequiredQualifications,       -- Job qualifications
    j.SalaryRange,                  -- Salary range

    u.ContactNo,  -- Company's contact number from User table
    j.create_at AS jobs_create_at,    -- Job creation timestamp
    c.create_at AS company_create_att -- Company creation timestamp
FROM 
    BookmarkJobs bj
INNER JOIN 
    Jobs j ON bj.JobID = j.JobID
INNER JOIN 
    company c ON j.CompanyID = c.CompanyID
INNER JOIN 
    user u ON c.CompanyID = u.UserID
ORDER BY 
    j.create_at DESC;
