	`RequiredQualifications` TEXT NULL DEFAULT NULL COLLATE 'utf8mb4_general_ci',
	`SalaryRange` DECIMAL(10,2) NOT NULL,

CREATE VIEW `v_jobs` AS SELECT 
    j.JobID AS JobID,
    c.CompanyID AS CompanyID,
    j.Title AS Title,
    c.CompanyLogo AS CompanyLogo,
    c.CompanyName AS CompanyName,
    i.IndustryName AS Industry,
    c.Website AS Website,
    c.Facebook,
    c.LinkedIn,
    j.Description AS Description,
    TRIM(REGEXP_REPLACE(CONCAT_WS(', ', c.StreetNo, c.AddressLine1, c.AddressLine2, ci.CityName), ', ,+', ',')) AS CompanyAddress,
    j.Category AS Category,
    j.Status AS Status,
    d.DistrictName AS District,
    ci.CityName AS City,
    j.JobBenefits AS JobBenefits,
    j.RequiredQualifications AS RequiredQualifications,
    j.SalaryRange AS SalaryRange,
    j.SalaryType AS SalaryType,
    j.PublishDate AS PublishDate,
    u.Email AS Email,
    u.ContactNo AS ContactNo,
    j.create_at AS jobs_create_at,
    c.create_at AS company_create_at
FROM 
    jobs j
    JOIN company c ON j.CompanyID = c.CompanyID
    JOIN user u ON c.CompanyID = u.UserID
    JOIN districts d ON j.DistrictID = d.DistrictID
    JOIN cities ci ON j.CityID = ci.CityID
    LEFT JOIN industry i ON c.IndustryID = i.IndustryID
ORDER BY 
    j.create_at DESC  ;

CREATE VIEW `v_company` AS SELECT 
    c.CompanyID,
    c.CompanyName,
    c.Description,
    c.CompanyLogo,
    CONCAT_WS(', ',
        CONCAT(c.StreetNo, ' ', c.AddressLine1),
        c.AddressLine2,
        ci.CityName
    ) AS Address,
    d.DistrictName AS District,
    ci.CityName AS City,
    i.IndustryName AS Industry,
    c.Website,
    c.Facebook,
    c.LinkedIn,
    u.Email,
    u.ContactNo,
    u.Role,
    u.Status,
    c.create_at AS company_create_at,
    (SELECT COUNT(*) FROM review r WHERE r.CompanyID = c.CompanyID) AS ReviewCount,
    (SELECT COUNT(*) FROM jobs j WHERE j.CompanyID = c.CompanyID AND j.Status = 'Active') AS ActiveJobsCount
FROM 
    company c
    JOIN user u ON c.CompanyID = u.UserID
    LEFT JOIN districts d ON c.DistrictID = d.DistrictID
    LEFT JOIN cities ci ON c.CityID = ci.CityID
    LEFT JOIN industry i ON c.IndustryID = i.IndustryID  ;
