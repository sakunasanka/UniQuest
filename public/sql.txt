CREATE VIEW v_jobs AS 
    SELECT 
        jobs.JobID AS JobID,
        company.CompanyID AS CompanyID,
        jobs.Title AS Title,
        company.CompanyLogo AS CompanyLogo,
        company.CompanyName AS CompanyName,
        jobs.Description AS Description,
        jobs.Location AS Location,
        jobs.Category AS Category,
        jobs.Status AS Status,
        jobs.JobBenefits AS JobBenefits,
        jobs.RequiredQualifications AS RequiredQualifications,
        jobs.SalaryRange AS SalaryRange,
        user.Email AS Email,
        user.ContactNo AS ContactNo,
        jobs.Create_at AS jobs_create_at,         -- Use correct capitalization
        company.Create_at AS company_create_at    -- Use correct capitalization
    FROM jobs 
    INNER JOIN company ON jobs.CompanyID = company.CompanyID
    INNER JOIN user ON company.CompanyID = user.UserID
    ORDER BY jobs.Create_at DESC;

-- -----------------------------------------------------
-- View uniquest.StudentCompanyComplaints
-- -----------------------------------------------------

CREATE VIEW uniquest.StudentCompanyComplaints AS
SELECT 
    c.CompanyName AS Company,
    u.Email AS CompanyEmail,
    cm.Complaint AS Complaint,
    CONCAT(s.FirstName, ' ', s.LastName) AS StudentName,
    cm.ComplainedDate AS ComplainedDate,
    cm.Status AS Status
FROM 
    uniquest.complaint_company cm
JOIN 
    uniquest.student s ON cm.StudentID = s.StudentID
JOIN 
    uniquest.company c ON cm.CompanyID = c.CompanyID
JOIN 
    uniquest.user u ON c.CompanyID = u.UserID 
ORDER BY 
    cm.ComplainedDate ASC;

-- View uniquest.v_bookmarkedJobs
-- -----------------------------------------------------

CREATE VIEW v_bookmarkedJobs AS
SELECT 
    bj.StudentID,                   -- From BookmarkJobs table
    j.JobID,                        -- From Jobs table
    c.CompanyID,                    -- From Company table
    j.Title,                        -- Job title
    c.CompanyLogo,                  -- Company logo
    c.CompanyName,                  -- Company name
    j.Description,                  -- Job description
    j.Location,                     -- Job location
    j.Category,                     -- Job category
    j.JobBenefits,                  -- Job benefits
    j.Status,
    j.RequiredQualifications,       -- Job qualifications
    j.SalaryRange,                  -- Salary range
    u.Email,        -- Company's email from User table
    u.ContactNo,  -- Company's contact number from User table
    j.create_at AS jobs_create_at,    -- Job creation timestamp
    c.create_at AS company_create_at, -- Company creation timestamp
    bj.bookmark_create_at
FROM 
    BookmarkJobs bj
INNER JOIN 
    Jobs j ON bj.JobID = j.JobID
INNER JOIN 
    company c ON j.CompanyID = c.CompanyID
INNER JOIN 
    user u ON c.CompanyID = u.UserID
ORDER BY 
    bj.bookmark_create_at DESC;

    complaint_jobs.ComplainedDate,
    complaint_jobs.Status

CREATE VIEW CompanyReviews AS
SELECT 
    c.CompanyID,
    c.CompanyName,
    c.Description AS CompanyDescription,
    c.CompanyLogo,
    r.ReviewID,
    r.Rating,
    r.Comment,
    r.StudentID,
    r.created_at,
    CONCAT(s.FirstName, ' ', s.LastName) AS StudentName

FROM 
    company c
JOIN 
    Review r
ON 
    c.CompanyID = r.CompanyID
JOIN 
    student s
ON 
    r.StudentID = s.StudentID;
